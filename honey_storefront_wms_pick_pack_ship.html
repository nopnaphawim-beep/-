DEPLOYMENT & INTEGRATION GUIDE

คู่มือการเชื่อมต่อ Google Sheets, GitHub และ Vercel

ทำตาม 3 ขั้นตอนด้านล่างเพื่อให้ระบบร้านค้าของคุณบันทึกออเดอร์ลง Google Sheets จริงและออนไลน์ผ่านโดเมนของ Vercel

1

เชื่อมต่อ Google Sheets (Web App URL)

* หากยังไม่ได้ใส่ URL ระบบจะทำงานในโหมดทดสอบ (บันทึกลง LocalStorage ในเบราว์เซอร์อัตโนมัติ)

โค้ด Google Apps Script (นำไปวางใน ส่วนขยาย > Apps Script ของ Google Sheets): 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Orders") || SpreadsheetApp.getActiveSpreadsheet().insertSheet("Orders");
  
  // สร้างหัวตารางอัตโนมัติหากยังไม่มีข้อมูล
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["OrderID", "Timestamp", "CustomerName", "Phone", "Address", "Zipcode", "PaymentMethod", "ItemsJSON", "ItemsSummary", "TotalAmount", "Status"]);
  }
  
  var data = JSON.parse(e.postData.contents);
  
  // กรณีอัปเดตสถานะจากระบบหลังบ้าน (Pick / Pack / Ship)
  if (data.action === "UPDATE_STATUS") {
    var rows = sheet.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] === data.orderId) {
        sheet.getRange(i + 1, 11).setValue(data.status);
        return ContentService.createTextOutput(JSON.stringify({ result: "updated" })).setMimeType(ContentService.MimeType.JSON);
      }
    }
  }
  
  // กรณีสร้างออเดอร์ใหม่จากหน้าร้าน
  sheet.appendRow([
    data.orderId,
    data.timestamp,
    data.customerName,
    "'" + data.phone, // ใส่ ' นำหน้าเพื่อให้เลข 0 ไม่หาย
    data.address,
    data.zipcode,
    data.paymentMethod,
    JSON.stringify(data.items),
    data.itemsSummary,
    data.totalAmount,
    data.status || "Pending Pick"
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ result: "success", orderId: data.orderId }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Orders");
  if (!sheet) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  
  var rows = sheet.getDataRange().getValues();
  var orders = [];
  for (var i = 1; i < rows.length; i++) {
    orders.push({
      orderId: rows[i][0],
      timestamp: rows[i][1],
      customerName: rows[i][2],
      phone: String(rows[i][3]).replace(/^'/, ''),
      address: rows[i][4],
      zipcode: rows[i][5],
      paymentMethod: rows[i][6],
      items: JSON.parse(rows[i][7] || "[]"),
      itemsSummary: rows[i][8],
      totalAmount: Number(rows[i][9]),
      status: rows[i][10] || "Pending Pick"
    });
  }
  return ContentService.createTextOutput(JSON.stringify(orders.reverse())).setMimeType(ContentService.MimeType.JSON);
}

วิธี Deploy ใน Apps Script: กดปุ่ม ทำให้ใช้งานได้ (Deploy) > รายการใหม่ (New deployment) > เลือกประเภทเป็น "เว็บแอป (Web app)" > ผู้มีสิทธิ์เข้าถึงเลือกเป็น "ทุกคน (Anyone)" แล้วคัดลอก URL มาวางด้านบน

2

โครงสร้างโฟลเดอร์สำหรับ GitHub (แยกโฟลเดอร์รูปภาพ)

สร้าง Repository ใหม่บน GitHub และจัดวางไฟล์ตามโครงสร้างนี้ เพื่อให้รูปภาพสินค้าแสดงผลได้อย่างถูกต้อง:

📁 my-honey-store/
├── 📄 index.html (ไฟล์โค้ดเว็บไซต์นี้ทั้งหมด)
└── 📁 public/
└── 📁 images/ (โฟลเดอร์เก็บรูปสินค้าแยกต่างหาก)
├── 🖼️ doikham-large.png (ดอยคำ ขวดใหญ่ 770g)
├── 🖼️ doikham-small.png (ดอยคำ ขวดเล็ก 230g)
├── 🖼️ doikham-tube.png (ดอยคำ หลอดบีบ 115g)
├── 🖼️ doikham-jar.png (ดอยคำ กระปุกแก้ว 135g)
├── 🖼️ vejpong-single.png (เวชพงศ์ 1 ขวด 300g)
└── 🖼️ vejpong-pack.png (เวชพงศ์ แพ็คคู่ 2 ขวด)

* หมายเหตุ: หากยังไม่ได้อัปโหลดรูปจริงลงโฟลเดอร์ public/images/ ระบบจะแสดงภาพกราฟิกขวดน้ำผึ้งจำลองที่สวยงามให้โดยอัตโนมัติ (Fallback)

3

นำขึ้นโดเมนของ Vercel

  1. เข้าสู่ระบบที่ vercel.com ด้วยบัญชี GitHub ของคุณ
  2. คลิกปุ่ม Add New... > Project และเลือก Repository my-honey-store
  3. ในหน้า Configure Project ตั้งค่า Output Directory เป็น . (หรือปล่อยค่าเริ่มต้นสำหรับ Static HTML)
  4. กดปุ่ม Deploy รอประมาณ 30 วินาที คุณจะได้ลิงก์โดเมน https://your-project.vercel.app พร้อมใช้งานทันที!
